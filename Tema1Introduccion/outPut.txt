package ejercicios1_1Directorios;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.Scanner;
import java.util.stream.Stream;

public class Ejercicio3 {
    /*
    Muestra el nombre y tipo (fichero o directorio) de los ficheros y
    subdirectorios contenidos en un directorio solicitado al usuario.
    Muestra también el contenido de todos los subdirectorios y si éstos contienen subdirectorios también...
    y así sucesivamente hasta mostrar todo el contenido de dicho directorio.
    No se puede utilizar el método walk de la clase Files.
    */
    public void show() {
        Scanner kb = new Scanner(System.in);
        Path pathToComprobe;
        System.out.println("Introduzca el path a combrobar");
        pathToComprobe = Paths.get(kb.nextLine());
        showFiles(pathToComprobe);
        showDirectories(pathToComprobe);
    }

    public void showFiles(Path userPath) {
        try (Stream<Path> filesContents = Files.list(userPath)) {
            Stream<Path> files = filesContents.filter(Files::isRegularFile);
            Iterable<Path> iterable = files::iterator;
            for (Path p : iterable) {
                System.out.printf("Archivo : %s\n", p);
            }
        } catch (IOException e) {
            System.err.printf("Error de Entrada/Salida", e.getMessage());
        }
    }

    public void showDirectories(Path userPath) {
        try (Stream<Path> filesContents = Files.list(userPath)) {
            Stream<Path> files = filesContents.filter(Files::isDirectory);
            Iterable<Path> iterable = files::iterator;
            for (Path p : iterable) {
                System.out.printf("Directorio : %s\n", p);
            }
        } catch (IOException e) {
            System.err.printf("Error de Entrada/Salida", e.getMessage());
        }
    }

    public static void main(String[] args) {

        new Ejercicio3().show();
    }
}

public class Ejercicio3 {
    /*
    Realiza este ejercicio utilizando las clases BufferedReader y BufferedWriter: concatena todos tus ficheros .java realizados hasta ahora en los ejercicios en uno solo utilizando código de programación. A continuación, lee dicho fichero y genera dos ficheros:

    Fichero donde estén las declaraciones realizadas en los paréntesis del try-with-resources, cada una en una línea y terminadas con ;. Ejemplo:


    Stream<Path> directoryContents;
    BufferedReader reader;
    BufferedWriter writer;
    Fichero donde estén las declaraciones de las distintas excepciones tratadas, cada una en una línea, terminadas con ; y sin repeticiones, es decir, la excepción que aparezca varias veces que solamente aparezca una vez en el fichero:


    FileNotFoundException e;
    IOException e;
    FileAlreadyExistsException e;
    NoSuchFileException e;
    NotDirectoryException e;

     */
    public void show() {
        Path dir = Paths.get(".");
        Path newFile = Paths.get("outPut.txt");
        Path decTry = Paths.get("tryWithResources.txt");
        Path decExc = Paths.get("exepciones.txt");
        Set<String> excepciones = new HashSet<>();
        Pattern patDec = Pattern.compile("([A-Za-z0-9_<>]+(?:<[A-Za-z0-9_<>]+>)?\\s+[A-Za-z0-9_]+)\\s*=");
        Pattern patCach = Pattern.compile("catch\\s*\\(([^)]+)\\)"); // catch (FileNotFoundException e) -> captura "FileNotFoundException e"
        Matcher matcher;

        boolean enTry = false;

        try (BufferedWriter bw = Files.newBufferedWriter(newFile)) {
            try (Stream<Path> content = Files.walk(dir)) {
                content.filter(p -> !Files.isDirectory(p) && p.toString().endsWith(".java"))
                        .forEach(archivo -> {
                                    try (BufferedReader br = Files.newBufferedReader(archivo)) {

                                        String line;
                                        bw.write("Archivo: " + archivo.getFileName().toString());
                                        bw.newLine();

                                        while ((line = br.readLine()) != null) {
                                            bw.write(line);
                                            bw.newLine();
                                        }
                                        bw.newLine();

                                    } catch (IOException e) {
                                        System.err.printf("Error leyendo el archivo: %s...\n%s", archivo.getFileName(), e.getMessage());
                                    }
                                }
                        );
                System.out.println("Fichero concatenado con exito");
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
        System.out.println("Fase 2: try-with-resources");
        try (BufferedReader br = Files.newBufferedReader(newFile);
             BufferedWriter bw = Files.newBufferedWriter(decTry)) {
            String line, lineTrim, clear;
            while ((line = br.readLine()) != null) {
                lineTrim = line.trim();
                if (lineTrim.contains("try") && lineTrim.contains("(")) {
                    enTry = true;
                }
                if (enTry) {
                    matcher = patDec.matcher(lineTrim);
                    while ((matcher.find())) {
                        clear = matcher.group(1).trim();
                        bw.write(clear + ";");
                        bw.newLine();
                    }
                    if (lineTrim.contains(")")) {
                        enTry = false;
                    }
                }

            }
            System.out.println("Fichero de declaraciones creado con exito!");
        } catch (IOException e) {
            System.err.printf("Error en la fase 2: %s", e.getMessage());
        }
        System.out.println("Fase 3, Excepciones");
        try (BufferedReader br = Files.newBufferedReader(newFile)) {
            String line, ctch, excClear;
            String[] part;
            int latSpace;
            while ((line = br.readLine()) != null) {
                matcher = patCach.matcher(line);
                while (matcher.find()) {
                    ctch = matcher.group(1).trim();
                    part = ctch.split("\\|");
                    for (String p : part) {
                        //Separo el nombre de la excepcion del nombre de la variable 'e', ya que generalmente estan separados por espacio
                        latSpace = p.lastIndexOf(' ');
                        excClear = (latSpace != -1) ? p.substring(0, latSpace).trim() : p;

                        if (!excClear.isEmpty()) {
                            excepciones.add(excClear + " e;"); //Para evitar duplicados
                        }
                    }

                }
            }
        } catch (IOException e) {
            System.err.printf("Error leyendo para las excepciones : %s", e.getMessage());
        }
        //Escribo el conjunto de excepciones únicas
        try (BufferedWriter bw = Files.newBufferedWriter(decExc)) {
            for (String e : excepciones) {
                bw.write(e);
                bw.newLine();
            }
            System.out.println("Fichero de excepciones creado con exito!");
        } catch (IOException e) {
            System.err.printf("Error escribiendo el fichero : %s", e.getMessage());
        }
    }

import java.io.BufferedReader;

public class Ejercicio2 {
    /*
    Realiza un programa para encriptar y desencriptar los datos de un fichero de texto.
     Se introduce una cadena por teclado que será la clave a aplicar para la encriptación y desencriptación.
     A cada carácter del fichero de texto original se le sumará una letra de la clave (el código Unicode),
     cuando se hayan acabado las letras de la palabra clave y aún no se hayan acabado los caracteres del fichero,
      se volverá al principio de la cadena para seguir aplicando la encriptación.
      Los datos encriptados se escribirán en un fichero destino, que será usado como origen para desencriptar.
      Para desencriptar se aplicará la fórmula a la inversa. Por ejemplo, si el fichero origen contiene “abcdef”
      y la palabra clave es “rosa”, en el fichero destino se escribirán los siguientes caracteres: ”ÓÑÖÅ×Õ”.
      Es decir, Ó es el resultado de sumar a y r, Ñ es el resultado de sumar b y o, y así sucesivamente.
     */
    public void show() {
        Scanner kb = new Scanner(System.in);
        int choice, caracterLeido, caracterEncriptado, caracterDesencriptado;
        int i = 0; //Contador
        Path dir = Paths.get("file3");
        Path source = Paths.get("file3", "source.txt");
        Path destination = dir.resolve("destination.txt");
        Path desencriptado = dir.resolve("desencriptado.txt");
        System.out.println("¿Quieres encriptar, desencriptar? (1/2/)");
        String password;
        char letraClave;
        do {
            choice = kb.nextInt();
            kb.nextLine();
            if (choice != 1 && choice != 2) {
                System.out.println("Opción incorrecta, debes introducir entre 1 y 2");
            }
        } while (choice != 1 && choice != 2);

        do {
            System.out.println("Escribe una clave de encriptación");
            password = kb.nextLine();
        } while (password.isEmpty());

        if (choice == 1) {
            System.out.println("Encriptando archivo...");
            try (BufferedReader br = Files.newBufferedReader(source);
                 BufferedWriter bw = Files.newBufferedWriter(destination)) {
                while ((caracterLeido = br.read()) != -1) {
                    letraClave = password.charAt(i % password.length());
                    caracterEncriptado = caracterLeido + letraClave;
                    bw.write((char) caracterEncriptado);
                    i++;

                }
            } catch (Exception e) {
                System.out.printf("Error: %s", e.getMessage());
            }
        } else if (choice == 2) {
            System.out.println("Desencriptando archivo...");
            i = 0;
            try (BufferedReader br = Files.newBufferedReader(destination);
                 BufferedWriter bw = Files.newBufferedWriter(desencriptado)) {
                while ((caracterLeido = br.read()) != -1) {
                    letraClave = password.charAt(i % password.length());

                    caracterDesencriptado = caracterLeido - letraClave;

                    bw.write((char) caracterDesencriptado);
                    i++;
                }
                System.out.println("El archivo ha sido desencriptado correctamente");
            } catch (Exception e) {
                System.out.printf("Error: %s", e.getMessage());
            }
        }


    }

    public static void main(String[] args) {
        new Ejercicio2().show();
    }
}

Si el destino existe y appended es false: se sobrescribe su contenido y se copia el fichero origen con todo el contenido en una única operación. Utiliza el método write de Files. Muestra por consola el fichero destino.

    public void copyWriteFiles(Path source, Path destination, boolean appended) {
        try {
            if (Files.isDirectory(destination)) {
                if (Files.isDirectory(source)) {
                    throw new IllegalArgumentException("El origen no puede ser un directorio si el destino lo es.");
                } else {

                    Path newDirectory = destination.resolve(source.getFileName());
                    Files.copy(source, newDirectory, StandardCopyOption.REPLACE_EXISTING);
                    System.out.println(Files.readString(newDirectory));
                }
            } else {
                if (Files.isDirectory(source)) {
                    throw new IllegalArgumentException("El origen no puede ser un directorio si el destino es un fichero.");
                }
                if (!Files.exists(destination)) {


                    for (String linea : Files.readAllLines(source)) {
                        Files.writeString(
                                destination,
                                linea + System.lineSeparator(),
                                StandardOpenOption.CREATE,
                                StandardOpenOption.APPEND
                        );
                    }

                    System.out.println(Files.readString(destination));
                } else {


                    if (!appended) {

                        byte[] contenido = Files.readAllBytes(source);

                        Files.write(destination, contenido);

                        System.out.println(Files.readString(destination));
                    } else {

                        String contenido = Files.readString(source);

                        Files.writeString(
                                destination,
                                contenido,
                                StandardOpenOption.APPEND
                        );

                        System.out.println(Files.readString(destination));
                    }

                }
            }
        } catch (RuntimeException e) {
            throw new RuntimeException(e);
        } catch (IOException e) {
            System.err.printf("Error de E/S durante la operación con ficheros: %s", e.getMessage());
        }

    }

        new Ejercicio1().show();
    }
